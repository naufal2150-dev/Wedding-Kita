        import { getFirestore, collection, doc, addDoc, updateDoc, deleteDoc, onSnapshot } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";

        // Konfigurasi Firebase (Sudah terisi dengan project wedding-kita-2125d)
        const firebaseConfig = typeof __firebase_config !== 'undefined' 
            ? JSON.parse(__firebase_config) 
            : {
                apiKey: "AIzaSyAUzGRcf_HToAJULOsbD37xk2BARET9Abw",
                authDomain: "wedding-kita-2125d.firebaseapp.com",
                projectId: "wedding-kita-2125d",
                storageBucket: "wedding-kita-2125d.firebasestorage.app",
                messagingSenderId: "765080188480",
                appId: "1:765080188480:web:d710245092dc5c3703d1a5"
            };

        const appId = typeof __app_id !== 'undefined' ? __app_id : 'wedding-kita-2125d';
        const app = initializeApp(firebaseConfig);
        const auth = getAuth(app);
        const db = getFirestore(app);

        let currentUser = null;
        let allItems = [];
        let currentFilter = 'all';
        let currentCategory = 'Semua';
        let unsubscribeSnapshot = null;

        // Ambil atau buat Room Code dari URL (tanpa localStorage)
        function getInitialRoomCode() {
            let room = null;
            try {
                const urlParams = new URLSearchParams(window.location.search);
                room = urlParams.get('room');
            } catch (e) {}

            if (!room) {
                room = 'LOVE-' + Math.random().toString(36).substring(2, 6).toUpperCase();
            }

            syncUrlWithRoom(room.toUpperCase());
            return room.toUpperCase();
        }

        function syncUrlWithRoom(code) {
            try {
                if (window.location.protocol !== 'blob:') {
                    const newUrl = window.location.protocol + "//" + window.location.host + window.location.pathname + '?room=' + code;
                    window.history.replaceState({ path: newUrl }, '', newUrl);
                }
            } catch (e) {}
        }

        let roomCode = getInitialRoomCode();
        document.getElementById('currentRoomBadge').textContent = roomCode;
        document.getElementById('displayRoomCode').textContent = roomCode;

        // Format Rupiah
        function formatRupiah(angka) {
            return new Intl.NumberFormat('id-ID', {
                style: 'currency',
                currency: 'IDR',
                minimumFractionDigits: 0,
                maximumFractionDigits: 0
            }).format(angka || 0);
        }

        // Toast Notification
        window.showToast = function(message) {
            const toast = document.getElementById('toast');
            document.getElementById('toastMsg').textContent = message;
            toast.classList.remove('opacity-0', 'pointer-events-none');
            toast.classList.add('translate-y-2');
            setTimeout(() => {
                toast.classList.add('opacity-0', 'pointer-events-none');
                toast.classList.remove('translate-y-2');
            }, 2600);
        }

        // Autentikasi Firebase
        async function initApp() {
            try {
                if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
                    await signInWithCustomToken(auth, __initial_auth_token);
                } else {
                    await signInAnonymously(auth);
                }
            } catch (error) {
                console.error("Auth error:", error);
                document.getElementById('loadingState').innerHTML = `
                    <div class="bg-red-50 border border-red-200 text-red-700 p-4 rounded-2xl text-xs max-w-sm mx-auto">
                        <p class="font-bold mb-1">Gagal Terhubung ke Database</p>
                        <p>Pastikan <b>Anonymous Sign-in</b> dan <b>Firestore Database (Test Mode)</b> sudah diaktifkan di Firebase Console kamu.</p>
                    </div>
                `;
            }
        }

        onAuthStateChanged(auth, (user) => {
            if (user) {
                currentUser = user;
                listenToItems();
            }
        });

        // Real-time Listener Firestore
        function listenToItems() {
            if (!currentUser) return;
            if (unsubscribeSnapshot) unsubscribeSnapshot();

            document.getElementById('loadingState').classList.remove('hidden');
            
            const itemsRef = collection(db, 'artifacts', appId, 'public', 'data', `wedding_items_${roomCode}`);

            unsubscribeSnapshot = onSnapshot(itemsRef, (snapshot) => {
                document.getElementById('loadingState').classList.add('hidden');
                allItems = [];
                snapshot.forEach((doc) => {
                    allItems.push({ id: doc.id, ...doc.data() });
                });

                // Urutkan: Belum dibeli paling atas, lalu berdasarkan waktu terbaru
                allItems.sort((a, b) => {
                    if (a.completed === b.completed) {
                        return (b.createdAt || 0) - (a.createdAt || 0);
                    }
                    return a.completed ? 1 : -1;
                });

                updateSummary();
                renderItems();
            }, (error) => {
                console.error("Firestore error:", error);
                document.getElementById('loadingState').classList.add('hidden');
                showToast("Gagal memuat data. Cek aturan Firestore.");
            });
        }

        // Update Ringkasan Keuangan & Progress
        function updateSummary() {
            let totalEst = 0;
            let totalSpent = 0;
            let completedCount = 0;

            allItems.forEach(item => {
                const price = Number(item.price) || 0;
                totalEst += price;
                if (item.completed) {
                    totalSpent += price;
                    completedCount++;
                }
            });

            const totalRem = Math.max(0, totalEst - totalSpent);

            document.getElementById('totalSpent').textContent = formatRupiah(totalSpent);
            document.getElementById('totalEstimated').textContent = formatRupiah(totalEst);
            document.getElementById('totalRemaining').textContent = formatRupiah(totalRem);

            const totalCount = allItems.length;
            const percentage = totalCount === 0 ? 0 : Math.round((completedCount / totalCount) * 100);
            
            document.getElementById('progressText').textContent = `${percentage}% (${completedCount}/${totalCount})`;
            document.getElementById('progressBar').style.width = `${percentage}%`;
        }

        // Render Daftar Item
        window.renderItems = function() {
            const listEl = document.getElementById('itemsList');
            const emptyEl = document.getElementById('emptyState');
            const keyword = document.getElementById('searchInput').value.toLowerCase().trim();
            
            const filtered = allItems.filter(item => {
                const matchStatus = 
                    currentFilter === 'all' ? true :
                    currentFilter === 'pending' ? !item.completed :
                    item.completed;
                
                const matchCat = currentCategory === 'Semua' ? true : item.category === currentCategory;
                const matchSearch = !keyword || 
                    (item.name && item.name.toLowerCase().includes(keyword)) || 
                    (item.notes && item.notes.toLowerCase().includes(keyword));
                
                return matchStatus && matchCat && matchSearch;
            });

            if (filtered.length === 0) {
                listEl.innerHTML = '';
                emptyEl.classList.remove('hidden');
                return;
            }

            emptyEl.classList.add('hidden');
            listEl.innerHTML = filtered.map(item => {
                const picBadge = 
                    item.pic === 'Cowok' ? '<span class="bg-blue-50 text-blue-700 border border-blue-100 px-2 py-0.5 rounded-md text-[10px] font-semibold">🤵 Cowok</span>' :
                    item.pic === 'Cewek' ? '<span class="bg-pink-50 text-pink-700 border border-pink-100 px-2 py-0.5 rounded-md text-[10px] font-semibold">👰 Cewek</span>' :
                    '<span class="bg-sand text-espresso px-2 py-0.5 rounded-md text-[10px] font-semibold">💑 Berdua</span>';

                return `
                <div class="group bg-white rounded-2xl p-4 border ${item.completed ? 'border-emerald-200/70 bg-stone-50/80' : 'border-champagne/80 shadow-xs'} transition-all flex items-start gap-3.5">
                    <!-- Checklist Toggle Button -->
                    <button onclick="toggleComplete('${item.id}', ${item.completed})" 
                        class="mt-0.5 w-6 h-6 rounded-full border-2 flex items-center justify-center shrink-0 transition-all active:scale-90 ${item.completed ? 'bg-sage border-sage text-white' : 'border-stone-300 hover:border-terracotta text-transparent'}">
                        <i class="fa-solid fa-check text-[11px]"></i>
                    </button>

                    <!-- Content Area (Click to Edit) -->
                    <div class="flex-1 min-w-0 cursor-pointer" onclick="openEditModal('${item.id}')">
                        <div class="flex items-center gap-1.5 mb-1.5 flex-wrap">
                            <span class="text-[10px] font-bold uppercase tracking-wider text-stone-500 bg-cream border border-champagne/60 px-2 py-0.5 rounded-md">
                                ${escapeHtml(item.category)}
                            </span>
                            ${picBadge}
                        </div>
                        
                        <h3 class="text-sm font-bold leading-snug ${item.completed ? 'line-through text-stone-400' : 'text-espresso'}">
                            ${escapeHtml(item.name)}
                        </h3>
                        
                        ${item.notes ? `<p class="text-xs text-stone-500 mt-1 line-clamp-2">${escapeHtml(item.notes)}</p>` : ''}
                        
                        <div class="flex items-center justify-between mt-3 pt-2.5 border-t border-sand/80 gap-2 flex-wrap">
                            <span class="text-sm font-bold ${item.completed ? 'text-sage' : 'text-terracotta'}">
                                ${formatRupiah(item.price)}
                            </span>

                            ${item.link ? `
                                <a href="${escapeHtml(formatUrl(item.link))}" target="_blank" rel="noopener noreferrer" onclick="event.stopPropagation()" 
                                   class="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-xl bg-orange-50 hover:bg-orange-100 text-orange-600 border border-orange-200 text-[11px] font-bold transition-all active:scale-95">
                                    <i class="fa-solid fa-bag-shopping"></i> Lihat di Toko
                                </a>
                            ` : ''}
                        </div>
                    </div>

                    <!-- Delete Button -->
                    <button onclick="deleteItem('${item.id}')" class="text-stone-300 hover:text-red-500 p-1.5 -mr-1 transition-colors" title="Hapus Item">
                        <i class="fa-regular fa-trash-can text-sm"></i>
                    </button>
                </div>
                `;
            }).join('');
        }

        function escapeHtml(text) {
            if (!text) return '';
            const div = document.createElement('div');
            div.textContent = text;
            return div.innerHTML;
        }

        function formatUrl(url) {
            if (!url) return '';
            const trimmed = url.trim();
            return /^https?:\/\//i.test(trimmed) ? trimmed : 'https://' + trimmed;
        }

        window.setFilter = function(status) {
            currentFilter = status;
            ['all', 'pending', 'completed'].forEach(tab => {
                const el = document.getElementById(`tab-${tab}`);
                if (tab === status) {

            currentCategory = cat;
            const buttons = document.querySelectorAll('.cat-btn');
            buttons.forEach(btn => {
                if (btn.textContent.includes(cat) || (cat === 'Semua' && btn.textContent === 'Semua')) {
                    btn.className = "cat-btn shrink-0 px-3.5 py-1.5 rounded-full text-xs font-semibold bg-espresso text-white transition-all";

                    btn.className = "cat-btn shrink-0 px-3.5 py-1.5 rounded-full text-xs font-medium bg-white border border-champagne text-stone-600 hover:border-espresso transition-all";

        }

        // --- CRUD FIRESTORE OPERATIONS ---
        window.saveItem = async function(e) {
            e.preventDefault();
            if (!currentUser) return showToast('Menunggu koneksi server...');

            const id = document.getElementById('editItemId').value;
            const name = document.getElementById('itemName').value.trim();
            const category = document.getElementById('itemCategory').value;
            const price = Number(document.getElementById('itemPrice').value) || 0;
            const link = document.getElementById('itemLink').value.trim();
            const notes = document.getElementById('itemNotes').value.trim();
            const pic = document.querySelector('input[name="pic"]:checked').value;

            const saveBtn = document.getElementById('saveBtn');
            saveBtn.disabled = true;
            saveBtn.textContent = 'Menyimpan...';

            try {
                const itemsRef = collection(db, 'artifacts', appId, 'public', 'data', `wedding_items_${roomCode}`);
                
                if (id) {
                    const docRef = doc(db, 'artifacts', appId, 'public', 'data', `wedding_items_${roomCode}`, id);
                    await updateDoc(docRef, { name, category, price, link, notes, pic });
                    showToast('Item diperbarui');
                } else {
                    await addDoc(itemsRef, {
                        name,
                        category,
                        price,
                        link,
                        notes,
                        pic,
                        completed: false,
                        createdAt: Date.now()
                    });
                    showToast('Berhasil ditambahkan');
                }
                closeItemModal();
            } catch (error) {
                console.error("Save error:", error);
                showToast('Gagal menyimpan data');
            } finally {
                saveBtn.disabled = false;
                saveBtn.textContent = 'Simpan Item';
            }
        }

        window.toggleComplete = async function(id, currentStatus) {
            if (!currentUser) return;
            try {
                const docRef = doc(db, 'artifacts', appId, 'public', 'data', `wedding_items_${roomCode}`, id);
                await updateDoc(docRef, { completed: !currentStatus });
                showToast(!currentStatus ? 'Ditandai sudah dibeli!' : 'Dikembalikan ke belum dibeli');
            } catch (error) {
                console.error("Toggle error:", error);
                showToast('Gagal mengubah status');
            }
        }

        window.deleteItem = async function(id) {
            if (!currentUser) return;
            try {
                const docRef = doc(db, 'artifacts', appId, 'public', 'data', `wedding_items_${roomCode}`, id);
                await deleteDoc(docRef);
                showToast('Item dihapus');
            } catch (error) {
                console.error("Delete error:", error);

            }
        }

        window.exportToCSV = function() {
            if (allItems.length === 0) return showToast('Belum ada data untuk di-export');
            const headers = ['Nama Barang', 'Kategori', 'Harga (Rp)', 'PIC', 'Status', 'Link Toko', 'Catatan'];
            const rows = allItems.map(i => [
                `"${(i.name || '').replace(/"/g, '""')}"`,
                `"${i.category || ''}"`,
                i.price || 0,
                `"${i.pic || ''}"`,
                i.completed ? 'Sudah Dibeli' : 'Belum Dibeli',
                `"${(i.link || '').replace(/"/g, '""')}"`,
                `"${(i.notes || '').replace(/"/g, '""')}"`
            ]);

            const csvContent = "data:text/csv;charset=utf-8," + [headers.join(','), ...rows.map(e => e.join(','))].join('\n');
            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", `Wedding_List_${roomCode}.csv`);
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
            showToast('File Excel/CSV berhasil diunduh');
        }

        window.exportToJSON = function() {
            if (allItems.length === 0) return showToast('Belum ada data untuk di-backup');
            const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(allItems, null, 2));
            const link = document.createElement("a");
            link.setAttribute("href", dataStr);
            link.setAttribute("download", `Backup_Wedding_${roomCode}.json`);
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
            showToast('File backup (.json) diunduh');
        }

        window.importFromJSON = function(event) {
            const file = event.target.files[0];
            if (!file || !currentUser) return;

            const reader = new FileReader();
            reader.onload = async function(e) {
                try {
                    const importedItems = JSON.parse(e.target.result);
                    if (!Array.isArray(importedItems)) throw new Error('Format salah');
                    
                    const itemsRef = collection(db, 'artifacts', appId, 'public', 'data', `wedding_items_${roomCode}`);
                    for (const item of importedItems) {
                        await addDoc(itemsRef, {
                            name: item.name || 'Item',
                            category: item.category || 'Lainnya',
                            price: Number(item.price) || 0,
                            link: item.link || '',
                            notes: item.notes || '',
                            pic: item.pic || 'Berdua',
                            completed: Boolean(item.completed),
                            createdAt: Date.now()
                        });
                    }
                    showToast(`${importedItems.length} item berhasil dipulihkan!`);
                    closeRoomModal();
                } catch (err) {
                    showToast('File backup tidak valid');
                }
            };
            reader.readAsText(file);
        }
